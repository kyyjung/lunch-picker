  // fill globe with little colorful capsules for decoration

  const capsuleColors = ['#FF8C7A','#FFE49A','#B8E8D4','#D9CBFF','#FFB6C1','#A0D8FF'];

  const positions = [

    {t:'10%', l:'16%'}, {t:'6%', l:'50%'}, {t:'14%', l:'72%'},

    {t:'28%', l:'6%'},  {t:'26%', l:'34%'}, {t:'30%', l:'58%'}, {t:'22%', l:'80%'},

  ];


  const capsules = positions.map((p, i) => {

    const c = document.createElement('div');

    c.className = 'capsule';

    c.style.setProperty('--t', p.t);

    c.style.setProperty('--l', p.l);

    c.style.setProperty('--top-c', capsuleColors[i % capsuleColors.length]);

    c.style.setProperty('--bot-c', '#ffffff');

    globe.appendChild(c);

    return c;

  });


  // generate a smooth, meandering set of keyframes so each capsule

  // drifts and bumps around on its own path, like it's floating in water

  function driftKeyframes(amplitude){

    const steps = 5 + Math.floor(Math.random() * 3);

    const frames = [];

    for (let i = 0; i <= steps; i++){

      const x = (Math.random() - 0.5) * amplitude * 2;

      const y = (Math.random() - 0.5) * amplitude * 2;

      const r = (Math.random() - 0.5) * (amplitude * 3);

      frames.push({ transform: `translate(${x.toFixed(1)}px, ${y.toFixed(1)}px) rotate(${r.toFixed(1)}deg)` });

    }

    frames.push(frames[0]); // ease back near the start so it loops smoothly

    return frames;

  }


  function startIdleFloat(el){

    return el.animate(driftKeyframes(4), {

      duration: 2200 + Math.random() * 1800,

      iterations: Infinity,

      easing: 'ease-in-out'

    });

  }


  let idleAnims = capsules.map(c => startIdleFloat(c));


  // vigorous shaking used while the machine is "rolling"

  function shakeCapsule(el){

    return el.animate(driftKeyframes(14), {

      duration: 650,

      iterations: 1,

      easing: 'ease-in-out'

    });

  }


  function launchConfetti(){

    const colors = ['#FF8C7A','#FFE49A','#B8E8D4','#D9CBFF','#FFB6C1'];

    for(let i=0;i<26;i++){

      const el = document.createElement('div');

      el.className = 'confetti';

      el.style.left = Math.random()*100 + 'vw';

      el.style.background = colors[Math.floor(Math.random()*colors.length)];

      el.style.animationDuration = (1.6 + Math.random()*1.2) + 's';

      el.style.animationDelay = (Math.random()*0.3) + 's';

      document.body.appendChild(el);

      setTimeout(()=> el.remove(), 3200);

    }

  }


  btn.addEventListener('click', () => {

    capsules.forEach(c => {

      };

    });

      resultName.textContent = pick.name;

      resultSub.textContent = pick.sub;

      resultCard.classList.add('show');

      launchConfetti();

      btn.disabled = false;


      // settle back into the calm floating motion

      idleAnims = capsules.map(c => startIdleFloat(c));

    }, 700);

  });
